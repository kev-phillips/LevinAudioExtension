#include <dmsdk/sdk.h>
#include "raylib.h"  // Assuming r-audio is included here

// Lua binding to play a sound
static int lua_play_sound(lua_State* L) {
    const char* soundFilePath = luaL_checkstring(L, 1);  // Get sound file path from Lua stack
    Sound sound = LoadSound(soundFilePath);
    PlaySound(sound);
    return 0;
}

// Lua binding to stop a sound
static int lua_stop_sound(lua_State* L) {
    const char* soundFilePath = luaL_checkstring(L, 1);
    Sound sound = LoadSound(soundFilePath);
    StopSound(sound);
    return 0;
}

// Lua binding to set volume for a sound
static int lua_set_volume(lua_State* L) {
    const char* soundFilePath = luaL_checkstring(L, 1);
    float volume = luaL_checknumber(L, 2);
    Sound sound = LoadSound(soundFilePath);
    SetSoundVolume(sound, volume);
    return 0;
}

// Register the bindings with Lua
static const luaL_Reg raudio_methods[] = {
    {"play_sound", lua_play_sound},
    {"stop_sound", lua_stop_sound},
    {"set_volume", lua_set_volume},
    {"play_xm", lua_play_xm},
    {"stop_xm", lua_stop_xm},
    {"set_xm_volume", lua_set_xm_volume},
    {NULL, NULL}
};

static void LuaInit(lua_State* L) {
    luaL_register(L, "raudio", raudio_methods);
}

DM_DECLARE_EXTENSION(raudio, "raudio", 0, 0, LuaInit, 0, 0, 0)
